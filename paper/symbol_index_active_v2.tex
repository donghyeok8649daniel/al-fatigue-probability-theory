% Active manuscript v2 symbol index.
\section*{Active symbol index}

\begin{description}
\item[$a$] Physical local normal spacing; unit: m.
\item[$a_0$] Reference normal spacing; unit: m.
\item[$\lambda=a/a_0$] Dimensionless normal spacing.
\item[$\lambda_0$] Structural/prestress spacing label at the reference loading phase.
\item[$\Lambda(\lambda_0,t)$] Quasistatic stable-branch spacing reached from structural label $\lambda_0$ under the prescribed stress history.
\item[$\lambda_c$] Operational normal-instability boundary defined by $\phi''(\lambda_c)=0$.
\item[$m,n$] Repulsive and attractive exponents of the generalized Lennard--Jones shape, with $m>n>1$.
\item[$\phi(\lambda)$] Dimensionless generalized-Lennard--Jones normal interaction shape.
\item[$\phi',\phi''$] First and second derivatives of $\phi$ with respect to $\lambda$.
\item[$E$] Loading-axis elastic modulus used in the stress bridge; unit: Pa.
\item[$A_0$] Mechanical atomic/reference area used in the microscopic calibration; unit: m$^2$.
\item[$A_c$] Characteristic cohesive area participating coherently in one reduced normal-instability event; unit: m$^2$; not calibrated in the present work.
\item[$m_a$] Atomic/reference mass associated with the microscopic normal coordinate; unit: kg.
\item[$t_0=\sqrt{m_a a_0/(EA_0)}$] Microscopic normal inertial time scale; unit: s.
\item[$\sigma(t)$] Prescribed external normal stress history; unit: Pa.
\item[$q(t)=\sigma(t)/E$] Reduced normal traction.
\item[$q_{\rm ref}$] Reduced traction at the reference phase at which $P_0$ is specified.
\item[$q_r(\lambda_0)=\phi'(\lambda_0)-q_{\rm ref}$] Local residual conjugate bias used in the structural $P_0$ embedding closure.
\item[$q_c=\phi'(\lambda_c)$] Maximum stable reduced traction of the retained normal potential.
\item[$P_0(\lambda)$] Structural/prestress initial spacing density at the reference phase; not the instantaneous thermal displacement distribution.
\item[$P(\lambda,t)$] Nonabsorbing structural spacing density transported by the quasistatic mechanical map.
\item[$P_b(\lambda,t)$] Intact survivor subdensity; its total mass is the local survival probability.
\item[$S(t)$] Local survival probability, $S=\int P_b\,d\lambda$.
\item[$F_{\rm ci}(t)$] Cumulative local crack-initiation first-passage probability, $F_{\rm ci}=1-S$.
\item[$\Delta\psi_c(\lambda)$] Dimensionless effective-potential climb from a stable spacing to the operational boundary $\lambda_c$ at the same reduced traction.
\item[$\Delta G_c(\lambda)=EA_ca_0\Delta\psi_c$] Characteristic-domain energy climb to the operational first-passage boundary; unit: J.
\item[$k_B$] Boltzmann constant; unit: J K$^{-1}$.
\item[$T$] Absolute temperature; unit: K.
\item[$\nu_s(\lambda)=\sqrt{\phi''(\lambda)}/(2\pi t_0)$] Small-oscillation attempt frequency of the coherent normal coordinate; unit: s$^{-1}$.
\item[$k_c(\lambda,T;A_c)$] Positive-flux rare first-passage rate to the operational boundary; unit: s$^{-1}$.
\item[$T_f=1/f$] Fatigue loading period; unit: s.
\item[$f$] Fatigue loading frequency; unit: Hz.
\item[$\mathcal H_c(\lambda_0)$] Integrated one-cycle first-passage hazard for structural label $\lambda_0$.
\item[$S_N$] Local survival probability after $N$ loading cycles.
\item[$P_{\lambda,M}$] Exact finite-chain empirical spacing probability measure.
\item[$F_M(\lambda,c,\tau)$] Exact finite-chain empirical spacing-rate phase-space measure.
\item[$u(\lambda,\tau)$] Conditional mean dimensionless spacing rate in the microscopic projection.
\item[$\Theta(\lambda,\tau)$] Conditional variance of the dimensionless spacing rate.
\item[$C_3$] Third central conditional moment of spacing rate.
\item[$\Psi$] Conditional covariance $\operatorname{Cov}(c,\ddot\lambda\mid\lambda,\tau)$ in the exact moment hierarchy.
\item[$\mu_0$] Measure over complete initial microscopic chain states in the exact push-forward reference formulation.
\end{description}
